4//! Acquisition stamps for LiDAR frames.
5//!
6//! A frame is stamped with the sensor's own start-of-frame time when the
7//! sensor reports its clock synchronized to the host (PTP with the host or
8//! another system as grandmaster), and otherwise with the host receive time
9//! of the frame's first packet minus a configured sensor latency.
10//!
11//! A synchronized sensor time is only accepted when it is plausible against
12//! the host receive time of the packet that carried it. This rejects sensor
13//! clocks that report synchronization but are not in the host domain, such
14//! as a clock that is still slewing towards its grandmaster after a step.
15
16use std::{
17    fmt,
18    time::{Duration, SystemTime, UNIX_EPOCH},
19};
20use tracing::{info, warn};
21
22/// Largest accepted delay from a sensor timestamp to the host receive time
23/// of the packet carrying it. Measured delays are about 16 ms for the
24/// Robosense E1R start-of-sweep packet and about 3 ms for Ouster packets.
25pub const MAX_SENSOR_LATENCY: Duration = Duration::from_millis(100);
26
27/// Largest accepted lead of a sensor timestamp over the host receive time of
28/// the packet carrying it, allowing for residual PTP offset.
29pub const MAX_SENSOR_LEAD: Duration = Duration::from_millis(5);
30
31/// Clock that a frame stamp was taken from.
32#[derive(Clone, Copy, Debug, PartialEq, Eq)]
33pub enum StampSource {
34    /// The sensor clock, synchronized to the host by PTP.
35    SensorClock,
36    /// The host receive time of the frame's first packet, less the
37    /// configured sensor latency.
38    HostReceive,
39}
40
41impl fmt::Display for StampSource {
42    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
43        match self {
44            StampSource::SensorClock => write!(f, "sensor clock (PTP)"),
45            StampSource::HostReceive => write!(f, "host receive time"),
46        }
47    }
48}
49
50/// Chooses the stamp of each frame and logs when the choice changes.
51#[derive(Debug)]
52pub struct FrameStamper {
53    sensor: &'static str,
54    host_latency_ns: u64,
55    source: Option<StampSource>,
56    rejecting: bool,
57}
58
59impl FrameStamper {
60    /// Creates a stamper for the named sensor with no host latency.
61    pub fn new(sensor: &'static str) -> Self {
62        Self {
63            sensor,
64            host_latency_ns: 0,
65            source: None,
66            rejecting: false,
67        }
68    }
69
70    /// Sets the sensor latency subtracted from host receive times.
71    pub fn set_host_latency(&mut self, latency: Duration) {
72        self.host_latency_ns = u64::try_from(latency.as_nanos()).unwrap_or(u64::MAX);
73    }
74
75    /// Source of the most recent stamp, `None` before the first frame.
76    #[allow(dead_code)] // Used by library consumers and tests
77    pub fn source(&self) -> Option<StampSource> {
78        self.source
79    }
80
81    /// Returns the stamp of a frame in nanoseconds since the Unix epoch.
82    ///
83    /// `rx_ns` is the host receive time of the frame's first packet.
84    /// `sensor_ns` is the sensor's start-of-frame time, given only when the
85    /// sensor reports its clock synchronized to the host.
86    pub fn stamp(&mut self, rx_ns: u64, sensor_ns: Option<u64>) -> u64 {
87        let accepted = sensor_ns.filter(|&sensor_ns| plausible(rx_ns, sensor_ns));
88
89        match (sensor_ns, accepted) {
90            (Some(sensor_ns), None) if !self.rejecting => {
91                warn!(
92                    sensor = self.sensor,
93                    receive_minus_sensor_ms = (rx_ns as i128 - sensor_ns as i128) as f64 / 1e6,
94                    "sensor reports a synchronized clock that disagrees with the host receive time"
95                );
96                self.rejecting = true;
97            }
98            (Some(_), None) => {}
99            _ => self.rejecting = false,
100        }
101
102        let (stamp, source) = match accepted {
103            Some(sensor_ns) => (sensor_ns, StampSource::SensorClock),
104            None => (
105                rx_ns.saturating_sub(self.host_latency_ns),
106                StampSource::HostReceive,
107            ),
108        };
109
110        if self.source != Some(source) {
111            info!(
112                sensor = self.sensor,
113                source = %source,
114                host_latency_ms = self.host_latency_ns as f64 / 1e6,
115                "frame stamp source"
116            );
117            self.source = Some(source);
118        }
119
120        stamp
121    }
122}
123
124/// Whether a sensor timestamp is consistent with the host receive time of
125/// the packet that carried it.
126pub fn plausible(rx_ns: u64, sensor_ns: u64) -> bool {
127    let delay = rx_ns as i128 - sensor_ns as i128;
128    delay >= -(MAX_SENSOR_LEAD.as_nanos() as i128) && delay <= MAX_SENSOR_LATENCY.as_nanos() as i128
129}
130
131/// Nanoseconds since the Unix epoch, zero for instants before it.
132pub fn system_time_ns(time: SystemTime) -> u64 {
133    time.duration_since(UNIX_EPOCH)
134        .map_or(0, |d| u64::try_from(d.as_nanos()).unwrap_or(u64::MAX))
135}

137#[cfg(test)]
138mod tests {
139    use super::*;
140
141    const RX: u64 = 1_790_000_000_000_000_000;
142    const MS: u64 = 1_000_000;
143
144    #[test]
145    fn unsynchronized_sensor_uses_host_receive_time() {
146        let mut stamper = FrameStamper::new("test");
147        assert_eq!(stamper.stamp(RX, None), RX);
148        assert_eq!(stamper.source(), Some(StampSource::HostReceive));
149    }
150
151    #[test]
152    fn host_latency_is_subtracted_from_receive_time_only() {
153        let mut stamper = FrameStamper::new("test");
154        stamper.set_host_latency(Duration::from_millis(15));
155        assert_eq!(stamper.stamp(RX, None), RX - 15 * MS);
156        assert_eq!(stamper.stamp(RX, Some(RX - 16 * MS)), RX - 16 * MS);
157        assert_eq!(stamper.source(), Some(StampSource::SensorClock));
158    }
159
160    #[test]
161    fn synchronized_sensor_within_window_is_used() {
162        let mut stamper = FrameStamper::new("test");
163        for sensor in [RX - 99 * MS, RX - 16 * MS, RX, RX + 4 * MS] {
164            assert_eq!(stamper.stamp(RX, Some(sensor)), sensor);
165            assert_eq!(stamper.source(), Some(StampSource::SensorClock));
166        }
167    }
168
169    #[test]
170    fn implausible_sensor_time_falls_back_to_host() {
171        let mut stamper = FrameStamper::new("test");
172        // Too old: a sensor clock still slewing after a host step, or
173        // drifting in holdover.
174        assert_eq!(stamper.stamp(RX, Some(RX - 3_000 * MS)), RX);
175        assert_eq!(stamper.source(), Some(StampSource::HostReceive));
176        // Ahead of the host: a TAI clock read as UTC.
177        assert_eq!(stamper.stamp(RX, Some(RX + 37_000 * MS)), RX);
178        // Recovers once the sensor clock converges.
179        assert_eq!(stamper.stamp(RX, Some(RX - 3 * MS)), RX - 3 * MS);
180        assert_eq!(stamper.source(), Some(StampSource::SensorClock));
181    }
182
183    #[test]
184    fn host_latency_saturates_at_epoch() {
185        let mut stamper = FrameStamper::new("test");
186        stamper.set_host_latency(Duration::from_secs(10));
187        assert_eq!(stamper.stamp(5, None), 0);
188    }
189
190    #[test]
191    fn plausibility_window_bounds() {
192        let latency = MAX_SENSOR_LATENCY.as_nanos() as u64;
193        let lead = MAX_SENSOR_LEAD.as_nanos() as u64;
194        assert!(plausible(RX, RX - latency));
195        assert!(!plausible(RX, RX - latency - 1));
196        assert!(plausible(RX, RX + lead));
197        assert!(!plausible(RX, RX + lead + 1));
198    }
199
200    #[test]
201    fn system_time_ns_clamps_pre_epoch() {
202        assert_eq!(system_time_ns(UNIX_EPOCH), 0);
203        assert_eq!(system_time_ns(UNIX_EPOCH + Duration::from_nanos(RX)), RX);
204        assert_eq!(system_time_ns(UNIX_EPOCH - Duration::from_secs(1)), 0);
205    }
206}
