4//! Detection of host clock steps.
5//!
6//! A timerfd on `CLOCK_REALTIME` armed with `TFD_TIMER_CANCEL_ON_SET` is
7//! cancelled by the kernel the moment the clock is set discontinuously
8//! (`clock_settime`, `settimeofday`, or an `adjtimex` step such as chrony's
9//! `makestep`). Slews do not cancel it. The step size is measured as the
10//! change in `CLOCK_REALTIME - CLOCK_MONOTONIC`, which only a step alters.
11
12use std::{
13    io, mem,
14    os::fd::{AsRawFd, FromRawFd, OwnedFd},
15};
16
17/// Largest gap accepted between the two `CLOCK_REALTIME` reads that bracket
18/// a `CLOCK_MONOTONIC` read; preemption inside the bracket adds an error
19/// equal to the gap.
20const MAX_BRACKET_NS: i128 = 10_000;
21
22/// Brackets taken at most by [`realtime_minus_monotonic`].
23const BRACKET_ATTEMPTS: usize = 5;
24
25/// Far-future expiry for the watch timer, in seconds after arming.
26const WATCH_HORIZON_SECS: libc::time_t = 10 * 365 * 24 * 3600;
27
28/// Waits for steps of the host `CLOCK_REALTIME`.
29pub struct ClockStepWatcher {
30    fd: OwnedFd,
31    offset_ns: i128,
32}
33
34impl ClockStepWatcher {
35    /// Creates and arms the watch timer.
36    pub fn new() -> io::Result<Self> {
37        // SAFETY: plain system call; the returned descriptor is checked and
38        // owned below.
39        let fd = unsafe { libc::timerfd_create(libc::CLOCK_REALTIME, libc::TFD_CLOEXEC) };
40        if fd < 0 {
41            return Err(io::Error::last_os_error());
42        }
43        // SAFETY: fd is a freshly created descriptor owned by nothing else.
44        let fd = unsafe { OwnedFd::from_raw_fd(fd) };
45        let mut watcher = Self { fd, offset_ns: 0 };
46        watcher.arm()?;
47        Ok(watcher)
48    }
49
50    /// Arms the timer, then records the clock offset: a step before the
51    /// timer is armed shows in the recorded offset, and a step after it
52    /// cancels the timer, so none is missed.
53    fn arm(&mut self) -> io::Result<()> {
54        let now = clock_ns(libc::CLOCK_REALTIME);
55        let spec = libc::itimerspec {
56            it_interval: libc::timespec {
57                tv_sec: 0,
58                tv_nsec: 0,
59            },
60            it_value: libc::timespec {
61                tv_sec: (now / 1_000_000_000) as libc::time_t + WATCH_HORIZON_SECS,
62                tv_nsec: 0,
63            },
64        };
65        // SAFETY: fd is a valid timerfd and spec is a valid itimerspec.
66        let err = unsafe {
67            libc::timerfd_settime(
68                self.fd.as_raw_fd(),
69                libc::TFD_TIMER_ABSTIME | libc::TFD_TIMER_CANCEL_ON_SET,
70                &spec,
71                std::ptr::null_mut(),
72            )
73        };
74        if err != 0 {
75            return Err(io::Error::last_os_error());
76        }
77        self.offset_ns = realtime_minus_monotonic();
78        Ok(())
79    }
80
81    /// Blocks until `CLOCK_REALTIME` is set discontinuously and returns the
82    /// size of the step in nanoseconds (positive when the clock moved
83    /// forward). A set that leaves the clock where it was returns a step
84    /// near zero.
85    pub fn wait(&mut self) -> io::Result<i128> {
86        loop {
87            let mut expirations = 0u64;
88            // SAFETY: reads at most 8 bytes into a u64 owned by this frame.
89            let n = unsafe {
90                libc::read(
91                    self.fd.as_raw_fd(),
92                    &mut expirations as *mut u64 as *mut libc::c_void,
93                    mem::size_of::<u64>(),
94                )
95            };
96            if n < 0 {
97                // ECANCELED is the step notification; an expiry (after the
98                // far-future horizon) is handled the same way.
99                let err = io::Error::last_os_error();
100                match err.raw_os_error() {
101                    Some(libc::ECANCELED) => {}
102                    Some(libc::EINTR) => continue,
103                    _ => return Err(err),
104                }
105            }
106
107            let before = self.offset_ns;
108            self.arm()?;
109            return Ok(self.offset_ns - before);
110        }
111    }
112}
113
114/// Current `CLOCK_REALTIME - CLOCK_MONOTONIC` in nanoseconds.
115///
116/// The clocks cannot be read atomically, so the monotonic read is bracketed
117/// by two realtime reads and the midpoint is used. Preemption inside the
118/// bracket adds an error up to its width, so up to `BRACKET_ATTEMPTS`
119/// brackets are taken until one is narrower than `MAX_BRACKET_NS`, keeping
120/// the narrowest.
121pub fn realtime_minus_monotonic() -> i128 {
122    let mut best = (i128::MAX, 0);
123    for _ in 0..BRACKET_ATTEMPTS {
124        let rt0 = clock_ns(libc::CLOCK_REALTIME);
125        let mono = clock_ns(libc::CLOCK_MONOTONIC);
126        let rt1 = clock_ns(libc::CLOCK_REALTIME);
127        let width = rt1 - rt0;
128        if width < best.0 {
129            best = (width, (rt0 + rt1) / 2 - mono);
130        }
131        if width <= MAX_BRACKET_NS {
132            break;
133        }
134    }
135    best.1
136}
137
138fn clock_ns(clock: libc::clockid_t) -> i128 {
139    let mut ts = libc::timespec {
140        tv_sec: 0,
141        tv_nsec: 0,
142    };
143    // SAFETY: ts is a valid timespec owned by this frame.
144    unsafe { libc::clock_gettime(clock, &mut ts) };
145    ts.tv_sec as i128 * 1_000_000_000 + ts.tv_nsec as i128
146}

148#[cfg(test)]
149mod tests {
150    use super::*;
151
152    #[test]
153    fn offset_is_stable_without_steps() {
154        let a = realtime_minus_monotonic();
155        std::thread::sleep(std::time::Duration::from_millis(20));
156        let b = realtime_minus_monotonic();
157        // Only slews can move it here: well under a millisecond in 20 ms.
158        assert!((b - a).abs() < 1_000_000, "offset moved {} ns", b - a);
159    }
160
161    #[test]
162    fn watcher_arms() {
163        let watcher = ClockStepWatcher::new().unwrap();
164        assert!(watcher.offset_ns > 0);
165    }
166}
